cmake_minimum_required(VERSION 3.18...3.30)
project(v8_killer_native LANGUAGES C CXX)
include(FetchContent)
if(POLICY CMP0169)
  cmake_policy(SET CMP0169 OLD)
endif()
set(CMAKE_POLICY_VERSION_MINIMUM 3.5 CACHE STRING "")
set(DOBBY_ARCHIVE https://codeload.github.com/jmpews/Dobby/tar.gz/223aabced0431525c7d45196f9409fc505d58ac8)
if(DEFINED ENV{V8_KILLER_DOBBY_ARCHIVE})
  set(DOBBY_ARCHIVE "$ENV{V8_KILLER_DOBBY_ARCHIVE}")
endif()
FetchContent_Declare(dobby
  URL "${DOBBY_ARCHIVE}"
  URL_HASH SHA256=b6e5054dd75c54dfe5a7b9585722af344984de8fbf0e8456b45260cbb0b8b73f)
FetchContent_GetProperties(dobby)
if(NOT dobby_POPULATED)
  FetchContent_Populate(dobby)
endif()
include(patch.cmake)
set(DOBBY_GENERATE_SHARED OFF CACHE BOOL "" FORCE)
set(DOBBY_DEBUG OFF CACHE BOOL "" FORCE)
set(Plugin.SymbolResolver OFF CACHE BOOL "" FORCE)
set(Plugin.GlobalOffsetTableHook OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
add_subdirectory(${dobby_SOURCE_DIR} ${dobby_BINARY_DIR} EXCLUDE_FROM_ALL)
add_library(v8_killer_native STATIC bridge.cc)
add_dependencies(v8_killer_native dobby)
target_include_directories(v8_killer_native PRIVATE ${dobby_SOURCE_DIR}/include)
install(TARGETS v8_killer_native dobby ARCHIVE DESTINATION lib)
install(FILES ${dobby_SOURCE_DIR}/LICENSE DESTINATION share/licenses/dobby)

add_executable(v8_killer_abi_test EXCLUDE_FROM_ALL abi_test.cc)
target_link_libraries(v8_killer_abi_test PRIVATE v8_killer_native dobby ${CMAKE_DL_LIBS})
if(WIN32)
  target_link_libraries(v8_killer_abi_test PRIVATE psapi)
endif()
